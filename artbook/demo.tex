\section{Convergence}
\begin{definition}[Convergent sequence]
A sequence $(a_n)$ \emph{converges} to $L$ if for every $\varepsilon>0$
there is $N$ with $|a_n-L|<\varepsilon$ for all $n\ge N$.
\end{definition}
\begin{theorem}[Bolzano--Weierstrass]
Every bounded sequence of real numbers has a convergent subsequence.
\end{theorem}
\begin{proof}
Bisect the interval, and keep a half that holds infinitely many terms.
\end{proof}
\begin{example}
$a_n=1/n$ converges to $0$.
\end{example}
